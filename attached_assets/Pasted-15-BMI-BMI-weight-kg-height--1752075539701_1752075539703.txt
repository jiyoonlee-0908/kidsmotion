/* bmi.ts ------------------------------------------------- */
import { normalCDF } from "./stats"; // Φ 함수

// 1) 2017 KNGC LMS 테이블 일부 예시(M, L, S)
export const bmiLMS: Record<'M'|'F', Record<number, [number, number, number]>> = {
  M: {
    7: [−2.0547, 16.06, 0.0953],   // 만 7세 남아
    8: [−1.8573, 16.88, 0.0923],
    /* ...연령별 추가... */
  },
  F: {
    7: [−2.1763, 16.24, 0.0965],
    8: [−1.9735, 17.09, 0.0937],
  }
};

// 2) z-score & percentile 계산 ---------------------------
export const bmiZ = (bmi: number, sex: 'M'|'F', age: number): number => {
  const [L, M, S] = bmiLMS[sex][age];
  return ((Math.pow(bmi / M, L) - 1) / (L * S));
};

export const bmiPercentile = (z: number): number =>
  normalCDF(z) * 100;

// 3) 등급 매핑 ------------------------------------------
export const bmiLabel = (pct: number): '저체중'|'정상'|'과체중'|'비만' =>
  pct < 5 ? '저체중' : pct < 85 ? '정상' : pct < 95 ? '과체중' : '비만';
적용 방법

measurementData 저장 전 z-score·백분위·라벨을 함께 저장.

UI 카드에 “BMI 11.8 (23 P, 정상)” 형식으로 표시 → 보호자가 즉시 이해.

추후 파일럿 데이터가 쌓이면, 자체 LMS 재적합 테이블만 교체하면 코드 수정 없이 자동 업데이트.

핵심 설득 포인트
국가 성장차트와 동일 방식 → 의료진·학교에서도 바로 신뢰.

숫자+색상+라벨 3중 표기 → 보호자·투자자 모두 직관적.

데이터 축적 뒤 LMS만 교체 → 시스템 구조 변경 없이 정밀도 향상.

이렇게 변경하면 계산식은 표준 그대로 유지하면서, 결과 해석의 과학성과 신뢰도가 대폭 향상됩니다. 추가 질문이나 코드 통합 단계에서 도움이 필요하면 알려주세요!
