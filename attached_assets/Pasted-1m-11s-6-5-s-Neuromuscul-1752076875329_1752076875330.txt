정리
• 5 s만 “Peak”, 15 s 이상은 **“해당 구간 전체 평균”**으로 백분위 계산.
• Peak(1 s 또는 5 s)도 DB에 저장해두면 피로지수·스프린트 분석에 활용 가능하지만, 등급·백분위는 Mean 값으로 통일하면 보호자·투자자 설명이 쉽습니다.

/* ---------------- powerMetrics.ts ---------------- */

// 1. rollingMean: n-sec 창 최대 평균 파워 구하기
export const rollingMean = (arr: number[], windowSec: number, hz: number): number => {
  const win = windowSec * hz;
  let best = 0, sum = 0;
  for (let i = 0; i < arr.length; i++) {
    sum += arr[i];
    if (i >= win) sum -= arr[i - win];
    if (i >= win - 1) best = Math.max(best, sum / win);
  }
  return best;
};

// 2. testPower 함수
export const testPower = (
  data: number[],        // 1-Hz 파워 시계열
  test: '5s'|'15s'|'30s'|'60s'|'180s'|'360s'
): number => {
  const len = parseInt(test);        // 구간 길이(sec)
  return test === '5s'
    ? rollingMean(data, 1, 1)        // Peak 1-s
    : data.slice(0, len).reduce((a,b)=>a+b,0) / len; // Mean 전체
};
가정: 파워 데이터는 1 Hz로 저장. 센서 해상도가 4 Hz 이상이면 hz만 조정하면 됩니다.
